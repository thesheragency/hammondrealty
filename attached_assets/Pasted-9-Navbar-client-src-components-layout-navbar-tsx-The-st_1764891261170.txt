17. Scroll Reveal Hook (client/src/hooks/use-scroll-reveal.ts)
The GSAP hook that powers the animations.

import { useRef } from "react";
import { useGSAP } from "@gsap/react";
import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";
gsap.registerPlugin(ScrollTrigger);
export function useScrollReveal(ref: React.RefObject<HTMLElement | null>, selector = ".reveal-item", delay = 0) {
  useGSAP(() => {
    const items = ref.current?.querySelectorAll(selector);
    if (items && items.length > 0) {
      gsap.fromTo(items,
        { y: 60, opacity: 0 },
        {
          y: 0,
          opacity: 1,
          duration: 1,
          stagger: 0.15,
          delay: delay,
          ease: "power3.out",
          scrollTrigger: {
            trigger: ref.current,
            start: "top 85%",
            toggleActions: "play none none reverse"
          }
        }
      );
    }
  }, { scope: ref });
}
